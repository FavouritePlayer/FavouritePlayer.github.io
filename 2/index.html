def getBox(k):
    # kxk box filter, each entry is 1/k^2
    box_filter = np.ones((k, k)) / (k * k)
    return box_filter

def convolve4(img, kernel, mode="same"):
    # flip the kernel as required
    kernel = kernel[::-1, ::-1]
    kh, kw = kernel.shape
    height, width = img.shape
    output_height = height + kh - 1
    output_width = width + kw - 1
    output = np.zeros((output_height, output_width))
    # pad kh - 1 and kw - 1 zeros on every side
    img = np.pad(img, ((kh - 1, kh - 1), (kw - 1, kw - 1)), mode="constant", constant_values=0)
    for i in range(output_height):
        for j in range(output_width):
            for ki in range(kh):
                for kj in range(kw):
                    output[i, j] += img[i + ki, j + kj] * kernel[ki, kj]
    if mode == "full":
        return output
    top = (kh - 1) // 2
    left = (kw - 1) // 2
    return output[top:top + height, left:left + width]

def convolve2(img, kernel, mode="same"):
    kernel = kernel[::-1, ::-1]
    kh, kw = kernel.shape
    height, width = img.shape
    output_height = height + kh - 1
    output_width = width + kw - 1
    output = np.zeros((output_height, output_width))
    img = np.pad(img, ((kh - 1, kh - 1), (kw - 1, kw - 1)), mode="constant", constant_values=0)
    for i in range(output_height):
        for j in range(output_width):
            window = img[i:i + kh, j:j + kw]
            output[i, j] = np.sum(window * kernel)
    if mode == "full":
        return output
    top = (kh - 1) // 2
    left = (kw - 1) // 2
    return output[top:top + height, left:left + width]

def gaussian(s):
    ksize = 2 * int(np.ceil(3 * s)) + 1
    G1D = cv2.getGaussianKernel(ksize=ksize, sigma=s)
    return G1D @ G1D.T

def laplacian(s):
    G = gaussian(s)
    unit = np.zeros_like(G)
    unit[G.shape[0] // 2][G.shape[1] // 2] = 1
    return unit - G

Part 2.3: Gaussian and Laplacian Stacks

Stacks stay at the original resolution. There is no downsampling and no cv2.pyrDown. Level 0 is the image. Level i is the image blurred with σ = 2i−1 (σ = 1, 2, 4, 8, 16, 32). Laplacian level i is Gaussian i minus Gaussian i+1, and the last Laplacian level is the blurriest Gaussian so the levels sum back to the image. Laplacian figures are rescaled to [0, 1] because those bands are signed.

[Explain a stack versus a pyramid, and what each Gaussian and Laplacian level is showing on the apple and the orange. Mention that σ doubles each level to imitate a pyramid that would have halved the resolution.]

def get_gaussian_stack(img, n):
    # G0 = img, Gi = img blurred with sigma 2 ** (i - 1)
    stack = [img]
    for i in range(1, n):
        stack.append(conv_color(img, gaussian(2 ** (i - 1))))
    return stack

def get_laplacian_stack(img, n):
    g_stack = get_gaussian_stack(img, n)
    stack = []
    for i in range(n - 1):
        stack.append(g_stack[i] - g_stack[i + 1])
    stack.append(g_stack[-1])
    return stack
